#include "HX711.h"
#include <Wire.h>
#include <LiquidCrystal_I2C.h>

HX711 scale;

// LCD address is usually 0x27 or 0x3F
LiquidCrystal_I2C lcd(0x27, 16, 2);

#define DOUT 3
#define CLK 2

void setup() {
  Serial.begin(9600);

  // HX711 setup
  scale.begin(DOUT, CLK);
  scale.set_scale(); // calibrate later
  scale.tare();      // reset to 0

  // LCD setup
  lcd.init();
  lcd.backlight();

  lcd.setCursor(0, 0);
  lcd.print("Weight Scale");
  delay(2000);
  lcd.clear();
}

void loop() {
  float weight = scale.get_units(10);

  // Serial monitor
  Serial.print("Weight: ");
  Serial.print(weight);
  Serial.println(" kg");

  // LCD display
  lcd.setCursor(0, 0);
  lcd.print("Weight:");

  lcd.setCursor(0, 1);
  lcd.print(weight);
  lcd.print(" kg   "); // spaces clear old digits

  delay(500);
}
